\begin{frame}{Input contract: identity, source version and two clocks}

  \renewcommand{\arraystretch}{1.3}
  \small
  \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{.24\linewidth}>{\raggedright\arraybackslash}p{.27\linewidth}>{\raggedright\arraybackslash}X@{}}
    \toprule
    \textbf{Field} & \textbf{Illustrative value} & \textbf{Role in processing} \\
    \midrule
    \texttt{asset\_id} & \texttt{router-001} & Groups one configuration history \\
    \texttt{snapshot\_id} & \texttt{router-001-v11} & Detects conflicting reuse of an ID \\
    \texttt{version} & 11 & Selects the adjacent predecessor, v10 \\
    \texttt{event\_time} & 08:00 UTC & Source time for watermark and timers \\
    \texttt{ingest\_time} & 08:03 UTC & Collector acceptance time \\
    \texttt{hash} & SHA-256 of content & Checks inline content integrity \\
    \bottomrule
  \end{tabularx}
  \par\medskip
  {\small The schema accepts exactly one of \texttt{content} or \texttt{content\_uri}.\\
  The current diff processor handles inline text; URI content is not fetched.}


  \vnote{Khoảng 45 giây. Một message không chỉ có nội dung file mà phải có danh tính, phiên bản và thời gian. Asset ID gom lịch sử của một đối tượng; snapshot ID xác định một bản ghi nguồn. Version mới là cơ sở chọn cặp liền kề, không dùng offset hoặc thời gian nhận. Event time là thời điểm nguồn quan sát, còn ingest time là lúc collector nhận. Ví dụ lệch ba phút chỉ minh hoạ hai đồng hồ. SHA-256 kiểm tra nội dung inline; đây không phải chữ ký xác thực nguồn. Schema cho phép content hoặc URI nhưng processor diff hiện chỉ xử lý văn bản inline, không tự tải file từ URI.}
\end{frame}

\begin{frame}{Kafka: delivery position differs from source version}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{.43\textwidth}
      \textbf{Three input topics}\par\medskip
      \texttt{config.assets}: metadata\\[5pt]
      \texttt{config.events}: change audit\\[5pt]
      \texttt{config.snapshots}: content\par\medskip
      \textbf{Key:} \texttt{asset\_id}\\[5pt]
      Spark processes the snapshots topic.
    \end{column}
    \begin{column}{.53\textwidth}
      \textbf{Example: one snapshot-topic partition}\par\medskip
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{@{}lccc@{}}
        \toprule
        Offset & 40 & 41 & 42 \\
        Asset & R1 & R1 & R1 \\
        Version & v1 & \textcolor{MediumRed}{v3} & \textcolor{MediumBlue}{v2} \\
        \bottomrule
      \end{tabular}\par\medskip
      Offset 42 arrives later, but v2 belongs\\between v1 and v3 in source history.
    \end{column}
  \end{columns}
  \medskip
  \begin{block}{Why Kafka and state are both needed}
    Kafka retains a replayable log per partition. Spark reconstructs version relationships.\\
    Checkpointed progress resumes processing while source offsets remain available.
  \end{block}


  \vnote{Khoảng 50 giây. Producer gửi metadata asset, sự kiện audit và snapshot vào ba topic riêng. Kafka key là asset ID; Spark xử lý topic snapshots và không chờ event ở topic khác đến trước. Bảng bên phải minh hoạ cùng asset: offset 40, 41, 42 mang version 1, 3, 2. Kafka vẫn giữ đúng thứ tự log nhưng lịch sử nguồn phải là 1, 2, 3. Kafka tách producer khỏi processor và giữ log để replay; Spark mới giải quyết quan hệ phiên bản. Checkpoint lưu tiến độ đã xử lý; nếu Kafka đã xoá offset do retention thì checkpoint không tự tạo lại dữ liệu.}
\end{frame}

\begin{frame}{Spark: two stateful stages before output publication}

  \centering
  \includegraphics[width=\linewidth,height=3.45cm,keepaspectratio]{spark-stages.png}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}\small
      \textbf{Key 1: \texttt{snapshot\_id}}\\[4pt]
      Validate content hash and source binding.\\
      Same binding: suppress retry. Conflict: diagnostic.
    \end{column}
    \begin{column}{.48\textwidth}\small
      \textbf{Key 2: \texttt{asset\_id}}\\[4pt]
      Retain normalized versions + emitted-edge flags.\\
      Event-time timers and a version cap bound state.
    \end{column}
  \end{columns}


  \vnote{Khoảng 65 giây. Đi theo sơ đồ từ parse tới output. SQL parse và kiểm tra envelope, Kafka key, thời gian. Sau đó có hai phép group và hai stateful operator native của Spark. Operator đầu group theo snapshot ID để kiểm tra binding về asset, version, thời gian nguồn, format, hash và URI. Binding giống nhau là retry; binding khác là conflict, kể cả khi cùng ID xuất hiện ở asset khác. Ingest time không thuộc danh tính nguồn. Operator sau mới group theo asset, chuẩn hoá nội dung, giữ buffer version và cờ cặp nào đã phát. Watermark và timer event time cùng giới hạn số version giữ bộ nhớ có giới hạn. Raw Kafka đi nhánh riêng nên bản ghi bị business logic từ chối vẫn còn bằng chứng. State được Spark checkpoint, không chỉ là dictionary Python trên driver.}
\end{frame}

\begin{frame}{Version state: buffer gaps and emit each adjacent pair}

  \centering
  \includegraphics[width=.85\linewidth,height=2.45cm,keepaspectratio]{version-order.png}
  \renewcommand{\arraystretch}{1.2}
  {\small\begin{tabularx}{\linewidth}{@{}p{.15\linewidth}p{.25\linewidth}X@{}}
    \toprule
    \textbf{Input} & \textbf{Retained versions} & \textbf{New output} \\
    \midrule
    v1 & \{1\} & Baseline only; no predecessor \\
    v3 & \{1, 3\} & Hold the gap; no 1 $\rightarrow$ 3 comparison \\
    v2 & \{1, 2, 3\} & Emit 1 $\rightarrow$ 2 and 2 $\rightarrow$ 3; mark both edges \\
    Retry v2 & Unchanged & No new logical diff while identity/state is retained \\
    \bottomrule
  \end{tabularx}}


  \vnote{Khoảng 60 giây. Đây là một trace nhỏ của thuật toán, không chỉ là hình sắp xếp. Khi v1 đến, lưu baseline. V3 đến nhưng thiếu v2 thì giữ khoảng trống và chưa tạo diff 1 sang 3. Khi v2 đến trong thời hạn còn giữ state, tìm được cả hai hàng xóm nên tạo 1 sang 2 và 2 sang 3. Cờ emitted-edge ghi nhận hai cặp đã phát. Nếu gửi lại v2 với cùng binding trong state còn giữ thì không tạo diff mới. Nếu dữ liệu quá muộn hoặc version đã bị eviction thì không hứa phục hồi online vô hạn; raw vẫn là bằng chứng của những record đã fetch. Timer tiến theo watermark event time, không đơn thuần theo đồng hồ treo tường.}
\end{frame}

config-lake/
  raw/snapshots/

    \end{column}
    \begin{column}{.54\textwidth}\small
      \textbf{Raw}: binary key/value plus Kafka topic, partition, offset and timestamp.\par\medskip
      \textbf{Processed}: typed normalized snapshots, diffs, alerts and diagnostics; Snappy Parquet.\par\medskip

    \end{column}
  \end{columns}

  {\footnotesize\texttt{processed/snapshots/query\_id=.../batch\_id=.../year=2026/month=10/day=5/}}
  \begin{block}{Three different meanings of partition}
    \small Kafka: a log shard.\quad Spark: execution tasks.\quad Lake: date-based file paths.
  \end{block}


  \vnote{Khoảng 55 giây. MinIO cung cấp object storage tương thích S3, Parquet là định dạng file có schema theo cột. Raw giữ byte và provenance của record Kafka được fetch, processed giữ snapshot đã chuẩn hoá cùng diff, alert và diagnostic. Query ID và batch ID chia vùng ghi để nhận diện lần retry; date folder dùng năm tháng ngày UTC, riêng snapshot lấy event time. Không partition theo asset ID. Ba chữ partition rất dễ lẫn: Kafka là shard log, Spark là phân chia task khi thực thi, lake là tổ chức file theo ngày. Thiết kế thư mục không tự chứng minh query pruning hay tăng tốc; cần đo reader thực tế. Chỉ các file được manifest commit khai báo mới là lịch sử được chấp nhận.}
\end{frame}

\begin{frame}{Commit protocol: publish verified history before serving}

  \centering
  \includegraphics[width=\linewidth,height=3.4cm,keepaspectratio]{commit-order.png}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}\small
      \textbf{Manifest binds physical output}\\[4pt]
      File paths, byte sizes, SHA-256 and counts.\\
      Reader applies the typed dataset schema.
    \end{column}
    \begin{column}{.48\textwidth}\small
      \textbf{Retry checks the full typed multiset}\\[4pt]
      Values and row multiplicity must match.\\
      Equal row counts alone are insufficient.
    \end{column}
  \end{columns}


  \vnote{Khoảng 55 giây. Một batch đi qua sáu bước. Trước hết ghi Parquet rồi kiểm tra inventory file, kích thước, SHA-256, schema và row count. Sau đó publish manifest để reader biết tập file nào hoàn tất. Search và Kafka alerts chạy sau commit lake; callback thành công mới cho Spark tiến hành checkpoint progress. Khi retry batch đã commit, không chỉ so số lượng dòng mà so toàn bộ projection có kiểu dữ liệu theo multiset cả hai chiều, bao gồm số lần mỗi dòng xuất hiện. Nếu khớp mới tái sử dụng output; khác thì fail. Bảo đảm này không biến ba hệ thống MinIO, Elasticsearch và Kafka thành một transaction atomic.}

\begin{frame}{Failure recovery: what happens if a batch stops midway?}

  \renewcommand{\arraystretch}{1.65}
  {\small\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}p{.27\linewidth}>{\raggedright\arraybackslash}p{.30\linewidth}>{\raggedright\arraybackslash}X@{}}
    \toprule
    \textbf{Failure point} & \textbf{Visible state} & \textbf{Retry behavior} \\
    \midrule
    Before lake manifest & Partial files are not committed history & Readers exclude the incomplete batch \\
    After commit, during search & Lake is valid; search may be incomplete & Verify/reuse lake; index deterministic document IDs again \\
    After alerts, before progress commit & Some alerts may already be delivered & Reprocessing may append physical alert duplicates \\
    \bottomrule
  \end{tabularx}}
  \par\medskip
  \textbf{Spark checkpoint:} source progress and operator state.\\[5pt]
  \textbf{Lake manifest:} which verified data files are committed.


  \vnote{Khoảng 50 giây. Ba tình huống giúp giải thích cơ chế recovery cụ thể. Lỗi trước manifest thì file có thể tồn tại nhưng reader không coi là lịch sử hoàn tất. Lỗi ở Elasticsearch sau khi lake commit thì batch Spark thất bại; khi retry, code kiểm tra và dùng lại lake, rồi index lại bằng deterministic document ID. Nếu alerts đã đến Kafka mà tiến độ batch chưa commit, retry có thể gửi lại alert vật lý. Vì vậy không gọi toàn hệ thống exactly once. Cần phân biệt hai marker: checkpoint thuộc Spark để tiếp tục source progress và state, manifest thuộc lake để công bố file hoàn tất. Đây là hành vi code; lần recovery Kubernetes trên image integrity mới vẫn chưa hoàn tất, không lấy kết quả image cũ thay cho nó.}
\end{frame}
