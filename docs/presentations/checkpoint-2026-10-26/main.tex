\begin{frame}[fragile]{The problem: arrival order can change the answer}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.47\textwidth}
      \begin{block}{Router configuration: version 10}
\begin{lstlisting}
hostname router-001
logging enabled
telnet disabled
\end{lstlisting}
      \end{block}
    \end{column}
    \begin{column}{.47\textwidth}
      \begin{alertblock}{Version 11}
\begin{lstlisting}
hostname router-001
logging disabled
telnet enabled
\end{lstlisting}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{.5cm}
  \keyline{The real problem is pairing states, not storing files}
  Snapshots arrive as messages. Version 2 can arrive after version 3, so arrival order cannot define the transition.
  \par\medskip
  \textbf{ConfigStream must order and validate temporal state before it computes a diff.}
  \source{Illustrative input from the executable demo: processing/demo.py.}
  \vnote{Khoảng 45 giây. Mở đầu bằng ví dụ cụ thể: cùng một router nhưng phiên bản sau tắt logging và bật telnet. Đây là dữ liệu minh hoạ do nhóm tạo, không phải sự cố thực tế. Nếu chỉ giữ file hiện tại, ta không biết thay đổi xảy ra lúc nào và khác phiên bản trước ra sao. Nhóm muốn lưu lịch sử và chỉ ra phần thay đổi thay vì chỉ giữ trạng thái cuối cùng.}
\end{frame}

% 3 — Product intent
\begin{frame}{If temporal state is wrong, every downstream result is wrong}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.31\textwidth}
      \begin{alertblock}{Late arrival}
        v1, v3, v2 can produce v1 $\rightarrow$ v3 if arrival order is trusted.
      \end{alertblock}
    \end{column}
    \begin{column}{.31\textwidth}
      \begin{alertblock}{Missing state}
        A missing state can look like a deletion or a real configuration change.
      \end{alertblock}
    \end{column}
    \begin{column}{.31\textwidth}
      \begin{alertblock}{Uncommitted result}
        A projection can show a result before the authoritative history is committed.
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{.65cm}
  \keyline{Therefore: temporal state first, diff second, publication third.}
  Input snapshots $\rightarrow$ ordered state $\rightarrow$ normalized diff $\rightarrow$ committed history $\rightarrow$ query
  \source{Failure modes formalized in the state, lake, and domain acceptance documents.}
  \vnote{Khoảng 40 giây. Nếu người nghe chỉ nhìn kết quả cuối, ba lỗi này đều có thể trông hợp lệ. Sai thứ tự tạo sai cặp phiên bản. Thiếu state tạo ra thay đổi giả. Projection có thể hiện kết quả trước khi lịch sử chuẩn được commit. Vì vậy cần giữ identity, producer order, completeness và provenance cùng với kết quả.}
\end{frame}

\begin{frame}{A trustworthy history needs four guarantees}
  \begin{enumerate}\setlength{\itemsep}{.7em}
    \item \textbf{Identity} links every observation to an asset and source coordinate.
    \item \textbf{Producer order} determines adjacency; transport arrival order does not.
    \item \textbf{Completeness} prevents the system from inferring a change from missing input.
    \item \textbf{Recovery evidence} lets readers rebuild projections from committed history.
  \end{enumerate}
  \vspace{.4cm}
  \keyline{These guarantees define the solution before any technology is chosen.}
  \source{Common evidence contract, stateful processing policy, and lake-first publication boundary.}
  \vnote{Khoảng 35 giây. Đây là yêu cầu trước khi nói đến Kafka hay Spark. Mỗi quan sát phải có identity và tọa độ nguồn. Adjacency phải theo thứ tự của producer. Input thiếu thì phải báo incomplete thay vì suy ra deletion. Cuối cùng, projection phải dựng lại được từ lịch sử đã commit.}
\end{frame}

\begin{frame}{Four datasets, four meanings of change}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \begin{block}{Terraform / TerraDS: versions}
        Infrastructure code stored in Git.\par\medskip

      \vspace{.15cm}
      \begin{block}{RIS: routing events}

      \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{block}{RPKI: observed daily states}
        Observed route-authorization records.\par\medskip
        \textbf{Day 1 $\rightarrow$ day 2}: which members differ?
      \end{block}
      \vspace{.15cm}
      \begin{block}{DNS: measurements}
        Recorded DNS query and response data.\par\medskip
        \textbf{Name $\rightarrow$ IP}: what was observed, when?
      \end{block}
    \end{column}
  \end{columns}
  \source{Conceptual examples. Sources and accepted scopes: frontend/public/research/v1/catalog.json.}
  \vnote{Khoảng 65 giây. Nhóm dùng bốn loại dữ liệu có ý nghĩa thời gian khác nhau. Terraform từ TerraDS là các phiên bản file mã hạ tầng trong Git: so sánh để biết dòng nào đổi, không coi commit là lần triển khai. RPKI là tập bản ghi uỷ quyền định tuyến quan sát theo ngày; so sánh thành viên giữa hai ngày đầy đủ, không suy ra chính xác thao tác của quản trị viên. RIS ghi nhận sự kiện BGP, ví dụ quảng bá hoặc rút tuyến; hai sự kiện giống nội dung vẫn có thể là hai lần xảy ra khác nhau. DNS là phép đo truy vấn và phản hồi: thấy tên miền trả về địa chỉ nào tại thời điểm quan sát, không coi đó là toàn bộ trạng thái DNS. Ví dụ trên slide chỉ minh hoạ cách hiểu. Điểm chung là cần lưu lịch sử; cách định nghĩa thay đổi phải riêng cho từng loại.}
\end{frame}
