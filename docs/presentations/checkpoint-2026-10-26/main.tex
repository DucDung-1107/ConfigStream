% Four distinct temporal data types
\begin{frame}{Four datasets, four meanings of change}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \begin{block}{Terraform / TerraDS: versions}
        Infrastructure code stored in Git.\par\medskip
        \textbf{v1 $\rightarrow$ v2}: which lines changed?
      \end{block}
      \vspace{.15cm}
      \begin{block}{RIS: routing events}
        BGP announcements and withdrawals.\par\medskip
        \textbf{Announce $\rightarrow$ withdraw}: what happened?
      \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{block}{RPKI: observed daily states}
        Observed route-authorization records.\par\medskip
        \textbf{Day 1 $\rightarrow$ day 2}: which members differ?
      \end{block}
      \vspace{.15cm}
      \begin{block}{DNS: measurements}
        Recorded DNS query and response data.\par\medskip
        \textbf{Name $\rightarrow$ IP}: what was observed, when?
      \end{block}
    \end{column}
  \end{columns}
  \source{Conceptual examples. Sources and accepted scopes: frontend/public/research/v1/catalog.json.}
  \vnote{Khoảng 65 giây. Nhóm dùng bốn loại dữ liệu có ý nghĩa thời gian khác nhau. Terraform từ TerraDS là các phiên bản file mã hạ tầng trong Git: so sánh để biết dòng nào đổi, không coi commit là lần triển khai. RPKI là tập bản ghi uỷ quyền định tuyến quan sát theo ngày; so sánh thành viên giữa hai ngày đầy đủ, không suy ra chính xác thao tác của quản trị viên. RIS ghi nhận sự kiện BGP, ví dụ quảng bá hoặc rút tuyến; hai sự kiện giống nội dung vẫn có thể là hai lần xảy ra khác nhau. DNS là phép đo truy vấn và phản hồi: thấy tên miền trả về địa chỉ nào tại thời điểm quan sát, không coi đó là toàn bộ trạng thái DNS. Ví dụ trên slide chỉ minh hoạ cách hiểu. Điểm chung là cần lưu lịch sử; cách định nghĩa thay đổi phải riêng cho từng loại.}
\end{frame}

  \vnote{Khoảng 55 giây. Chỉ vào hàng trên: dữ liệu có thể đến theo thứ tự 1, 3, 2. Nếu so sánh theo thứ tự đến, ta sẽ tạo cặp 1 sang 3 và bỏ qua phiên bản 2. Hàng dưới là cách nhóm xử lý: giữ các phiên bản theo asset và chỉ tạo diff giữa các phiên bản liền kề. Nếu 2 đến khi state còn giữ được 1 và 3 thì tạo đủ hai cặp đúng. State có giới hạn lưu giữ, nên không hứa sửa được mọi dữ liệu đến muộn vô hạn.}
\end{frame}

\begin{frame}{System architecture: four separate processing paths}
  \centering
  \includegraphics[width=\linewidth,height=4.7cm,keepaspectratio]{four-paths.png}
  \begin{flushleft}\small
    Kafka + Spark + MinIO (Parquet), with separate schemas and processing.\\
    Config search: Elasticsearch + FastAPI. Dashboard: accepted results from all four.
  \end{flushleft}
  \source{spark/streaming/; spark/rpki\_stream.py; spark/ris.py; spark/dns.py; domain architecture docs.}
  \vnote{Khoảng 60 giây. Bốn hàng là bốn đường xử lý đã có trong repo, không phải tất cả đều đi qua bộ diff cấu hình. Hàng Terraform giữ logic phiên bản và rule. RPKI xử lý tập thành viên của các ngày đầy đủ và thay đổi giữa chúng. RIS giữ các lần xảy ra sự kiện, chỉ hoà giải bản gửi lại cùng danh tính nguồn; không xoá sự kiện chỉ vì nội dung giống nhau. DNS giữ phép đo và nhóm kết quả có thông tin nguồn để so sánh mô tả. Các luồng dùng chung công nghệ Kafka, Spark và lưu Parquet trên MinIO nhưng có schema, topic và metadata commit riêng. Elasticsearch và FastAPI phục vụ đường cấu hình; dashboard hiện có bản tổng hợp bằng chứng của cả bốn miền, không phải API live chung cho mọi miền. Dữ liệu thực ở đây được phát lại từ các nguồn đã cố định.}
\end{frame}
